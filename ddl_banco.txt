create table admin_user(
    id serial primary key,
	login varchar(50) not null unique,
	password varchar(100) not null
);

create table switch (
    id serial primary key,
    hostname varchar(100) not null,
    ipv4 varchar(15) not null unique,
);

create table port (
    id serial primary key,
    switch_id integer not null references switch(id) on delete cascade,
    number integer not null,                    -- número da porta física
    ipv4 varchar(15),                           -- se a porta for gerenciável
    status boolean not null default true,       -- true = ativa, false = bloqueada
    lockable boolean not null default false,  -- portas que não podem ser desligadas
    unique (switch_id, number)                  -- evita duplicar portas no mesmo switch
);

CREATE TABLE trap_event (
    id serial PRIMARY KEY,
    switch_id integer NOT NULL REFERENCES switch(id) ON DELETE CASCADE,
    port_id integer NOT NULL REFERENCES port(id) ON DELETE CASCADE,
    event_type varchar(20) NOT NULL,   -- 'link_up' ou 'link_down'
    event_time timestamp NOT NULL DEFAULT now(),
    raw_data text
);
