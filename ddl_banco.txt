create table admin_user (
    id serial primary key,
    login varchar(50) not null unique,
    password varchar(100) not null,
    mac varchar(17) not null
);

create table switch (
    id serial primary key,
    hostname varchar(100) not null,
    ipv4 varchar(15) not null unique
);

create table port (
    id serial primary key,
    ifindex integer,
    hostname varchar(100),
    switch_id integer not null references switch(id) on delete cascade,
    number integer not null,
    ipv4 varchar(15),
    is_blocked boolean not null default true,
    lockable boolean not null default false,
    mac varchar(17),
    unique (switch_id, number)
);

create table scheduled_port_tasks (
    id serial primary key,
    port_id integer not null references port(id) on delete cascade,
    action varchar(10) not null check (action in ('block', 'unblock')),
    execute_at timestamp not null,
    executed boolean default false,
    executed_at timestamp
);
